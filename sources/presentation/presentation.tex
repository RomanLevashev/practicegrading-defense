\begin{frame}{PracticeGrading и исходные проблемы}
  \textbf{Назначение системы}

  PracticeGrading используется при проведении защит учебных практик
  на кафедре системного программирования СПбГУ.
  Система хранит сведения о студентах и работах, принимает оценки
  членов комиссии и рассчитывает рекомендуемый результат.

  \vspace{0.8em}
  \textbf{Исходные проблемы}

  \begin{itemize}
    \item Нового администратора приходилось добавлять вручную
      через базу данных
    \item Участник мог представиться другим членом комиссии
      и выставлять оценки от его имени
    \item Обновление приложения на рабочем сервере выполнялось вручную
      и зависело от действий администратора сервера
  \end{itemize}
\end{frame}
\begin{frame}{Аналоги и выбранный подход}
  \begin{itemize}
    \item В СПбГУ доступна Blackboard Learn; Moodle и Canvas также
      поддерживают задания, критерии и электронный журнал
    \item Эти платформы ориентированы на универсальный учебный курс
    \item Кафедральный процесс требует заседаний с составом комиссии,
      независимых оценок, специального расчёта и документов
    \item Поэтому выбрано развитие PracticeGrading, уже учитывающего
      принятый на кафедре процесс защиты
  \end{itemize}

\begin{frame}{Постановка задачи}
  \textbf{Цель работы} --- повысить безопасность PracticeGrading
  и автоматизировать выпуск новых версий.

  \vspace{1em}
  \textbf{Задачи}
  \begin{itemize}
    \item Добавить создание администраторов и смену пароля
    \item Реализовать безопасный доступ членов комиссии к оцениванию
    \item Реализовать Continuous Deployment
    \item Настроить ежедневное резервное копирование базы данных
    \item Добавить автоматические тесты
  \end{itemize}
\end{frame}

\begin{frame}{Управление администраторами}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \textbf{Создание администратора}
      \begin{itemize}
        \item Доступно авторизованному администратору
        \item Учётная запись создаётся через интерфейс приложения
        \item Ручное редактирование базы не требуется
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Смена пароля}
      \begin{itemize}
        \item Пользователь указывает текущий пароль
        \item Сервер проверяет новый пароль и сохраняет его хеш
        \item Ошибки возвращаются без изменения учётной записи
      \end{itemize}
    \end{column}
  \end{columns}

  \vfill
  \centering
  Операции защищены серверной проверкой роли пользователя
\end{frame}

\begin{frame}{Доступ членов комиссии}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.47\textwidth}
      \textbf{Подтверждаемый доступ}
      \begin{enumerate}
        \item Участник открывает ссылку на заседание
        \item Отправляет запрос от своего имени
        \item Запрос подтверждает участник с доступом
        \item Участник получает право выставлять оценки
      \end{enumerate}
    \end{column}
    \begin{column}{0.47\textwidth}
      \textbf{Доверенный доступ}
      \begin{enumerate}
        \item Администратор выдаёт персональную ссылку
        \item Участник один раз активирует доступ
        \item Ссылки на заседания открываются без подтверждения
        \item При необходимости участник добавляется в комиссию автоматически
      \end{enumerate}
    \end{column}
  \end{columns}

  \vfill
  \centering
  Доверенный доступ можно перевыпустить или отозвать.
  Старый токен перестаёт действовать
\end{frame}

\begin{frame}{Развёртывание и полный откат}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \textbf{Успешный выпуск}
      \begin{enumerate}
        \item Включается режим обслуживания
        \item Создаётся копия базы для отката
        \item Применяются новые SQL-миграции
        \item Запускается новый образ API
        \item После liveness-проверки переключается frontend
      \end{enumerate}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Ошибка в изменяющей части}
      \begin{enumerate}
        \item Останавливается новая версия API
        \item Восстанавливается база из созданной копии
        \item Возвращаются прежние образ API и frontend
        \item Проверяется доступность приложения
        \item При успешном откате снимается режим обслуживания
      \end{enumerate}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Резервное копирование}
  \begin{itemize}
    \item Systemd ежедневно запускает скрипт резервного копирования
    \item Скрипт создаёт дамп базы данных с помощью \texttt{pg\_dump}
    \item Целостность копии проверяется по контрольной сумме
    \item Копии старше 14 дней автоматически удаляются
    \item Сейчас копии хранятся на рабочем сервере
  \end{itemize}
\end{frame}

\begin{frame}{Тестирование}
  \begin{itemize}
    \item Добавлены backend-тесты на NUnit:
      создание администратора, смена пароля, выдача и отзыв доступа,
      проверка защищённых endpoints
    \item Добавлены сквозные сценарии Playwright:
      запрос и подтверждение обычного доступа, активация доверенной ссылки,
      вход без подтверждения и блокировка после отзыва
    \item CD проверен в окружении, повторяющем состав production
      \begin{itemize}
        \item поздняя ошибка после изменения базы, API и frontend
          завершилась полным откатом
        \item обычный выпуск завершился активацией новой версии
          и успешной liveness-проверкой
      \end{itemize}
    \item На проде успешно выполнены развёртывание и первый ежедневный бэкап
    \item Вручную проверены доступ, оценки и генерация документов
  \end{itemize}
\end{frame}

\begin{frame}{Результаты}
  \begin{itemize}
    \item Реализованы создание администратора и безопасная смена пароля
    \item Добавлены подтверждаемый и доверенный доступ членов комиссии
    \item Подготовлены версионированные SQL-миграции
    \item Реализован CD для API, frontend и базы данных
    \item Проверен полный откат после поздней ошибки развёртывания
    \item CD запущен на проде, настроены ежедневные бэкапы с хранением 14 дней
  \end{itemize}
